# Native static ICU4C build, including data compiled from this exact checkout.
cmake_minimum_required(VERSION 3.21)
if(TARGET icu)
    return()
endif()
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "This ICU adapter currently requires a native Windows/Linux build to run its data tools.")
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(icu_source "../../icu4c/source" ABSOLUTE)
set(icu_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${icu_generated}")
file(STRINGS "${icu_source}/common/unicode/uvernum.h" icu_version_line
    REGEX "^#define U_ICU_VERSION_MAJOR_NUM ")
string(REGEX MATCH "[0-9]+$" icu_major "${icu_version_line}")
if(NOT icu_major)
    message(FATAL_ERROR "Cannot determine ICU data version from uvernum.h.")
endif()

# Read upstream's maintained source manifests, rather than collecting test files.
function(icu_sources output directory)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${icu_source}/${directory}/sources.txt")
    file(STRINGS "${icu_source}/${directory}/sources.txt" sources)
    list(FILTER sources EXCLUDE REGEX "^(#|[ \t]*$)")
    list(TRANSFORM sources PREPEND "${icu_source}/${directory}/")
    set(${output} ${sources} PARENT_SCOPE)
endfunction()

# Register static modules with fips when imported; also support plain CMake tests.
function(icu_static target)
    if(COMMAND fips_begin_lib)
        set(BUILD_SHARED_LIBS OFF)
        fips_begin_lib(${target})
            target_sources(${target} PRIVATE ${ARGN})
        fips_end_lib()
    else()
        add_library(${target} STATIC ${ARGN})
    endif()
endfunction()
icu_sources(common_sources common)
icu_sources(i18n_sources i18n)
add_library(icu_common_objects OBJECT ${common_sources})
add_library(icu_i18n_objects OBJECT ${i18n_sources})
foreach(target icu_common_objects icu_i18n_objects)
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_include_directories(${target} PRIVATE "${icu_source}/common" "${icu_source}/i18n")
    target_compile_definitions(${target} PRIVATE U_STATIC_IMPLEMENTATION)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHsc /GR /utf-8)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -fexceptions -frtti)
    endif()
endforeach()
target_compile_definitions(icu_common_objects PRIVATE U_COMMON_IMPLEMENTATION)
target_compile_definitions(icu_i18n_objects PRIVATE U_I18N_IMPLEMENTATION)
# Upstream mixes its legacy UBool with bool; fips MSVC treats this warning as an error.
if(MSVC)
    target_compile_options(icu_i18n_objects PRIVATE /wd4805)
endif()

# Bootstrap tools use empty data. Final libraries use the generated complete data.
icu_static(icu_stubdata "${icu_source}/stubdata/stubdata.cpp")
target_compile_features(icu_stubdata PRIVATE cxx_std_17)
target_compile_definitions(icu_stubdata PRIVATE U_STATIC_IMPLEMENTATION)
target_include_directories(icu_stubdata PRIVATE "${icu_source}/common")
icu_static(icu_common_bootstrap $<TARGET_OBJECTS:icu_common_objects>)
target_link_libraries(icu_common_bootstrap PUBLIC icu_stubdata Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(icu_common_bootstrap PUBLIC advapi32)
endif()
icu_static(icu_i18n_bootstrap $<TARGET_OBJECTS:icu_i18n_objects>)
target_link_libraries(icu_i18n_bootstrap PUBLIC icu_common_bootstrap)

icu_sources(tool_sources tools/toolutil)
icu_static(icu_toolutil ${tool_sources})
target_compile_features(icu_toolutil PUBLIC cxx_std_17)
target_include_directories(icu_toolutil PUBLIC
    "${icu_source}/common" "${icu_source}/i18n" "${icu_source}/tools/toolutil")
target_compile_definitions(icu_toolutil PUBLIC U_STATIC_IMPLEMENTATION PRIVATE U_TOOLUTIL_IMPLEMENTATION)
target_link_libraries(icu_toolutil PUBLIC icu_i18n_bootstrap)
if(MSVC)
    # Upstream diagnostic printf uses unsigned long for size_t on Windows.
    target_compile_options(icu_toolutil PRIVATE /EHsc /GR /utf-8 /wd4805 /wd4477)
else()
    target_compile_options(icu_toolutil PRIVATE -fexceptions -frtti)
endif()

set(icu_tools genbrk genccode gencfu gencnval gendict genrb gensprep icupkg makeconv)
foreach(tool IN LISTS icu_tools)
    if(EXISTS "${icu_source}/tools/${tool}/sources.txt")
        icu_sources(tool_sources "tools/${tool}")
    else()
        file(GLOB tool_sources CONFIGURE_DEPENDS
            "${icu_source}/tools/${tool}/*.c" "${icu_source}/tools/${tool}/*.cpp")
    endif()
    add_executable(icu_${tool} ${tool_sources})
    target_link_libraries(icu_${tool} PRIVATE icu_toolutil)
    set_target_properties(icu_${tool} PROPERTIES
        OUTPUT_NAME "${tool}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tools/${tool}/$<CONFIG>")
    if(MSVC)
        target_compile_options(icu_${tool} PRIVATE /EHsc /GR /utf-8 /wd4805)
    else()
        target_compile_options(icu_${tool} PRIVATE -fexceptions -frtti)
    endif()
    list(APPEND icu_tool_targets icu_${tool})
endforeach()

# Use a C source data image for portability; no runtime DLL or .dat deployment.
file(GLOB_RECURSE icu_data_inputs CONFIGURE_DEPENDS "${icu_source}/data/*" "${icu_source}/python/*.py")
set(icu_data_c "${icu_generated}/$<CONFIG>/icudata.c")
add_custom_command(OUTPUT "${icu_data_c}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build_data.py"
        --source "${icu_source}" --output "${icu_generated}/$<CONFIG>"
        --tools "${CMAKE_CURRENT_BINARY_DIR}/tools" --config "$<CONFIG>" --major "${icu_major}"
    DEPENDS ${icu_tool_targets} ${icu_data_inputs} "${CMAKE_CURRENT_SOURCE_DIR}/build_data.py"
    VERBATIM)
icu_static(icudata "${icu_data_c}")
target_include_directories(icudata PRIVATE "${icu_source}/common")
target_compile_definitions(icudata PRIVATE U_STATIC_IMPLEMENTATION)

icu_static(icuuc $<TARGET_OBJECTS:icu_common_objects>)
target_include_directories(icuuc PUBLIC "${icu_source}/common")
target_compile_definitions(icuuc PUBLIC U_STATIC_IMPLEMENTATION)
target_compile_features(icuuc PUBLIC cxx_std_17)
target_link_libraries(icuuc PUBLIC icudata Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(icuuc PUBLIC advapi32)
endif()
icu_static(icui18n $<TARGET_OBJECTS:icu_i18n_objects>)
target_include_directories(icui18n PUBLIC "${icu_source}/i18n")
target_link_libraries(icui18n PUBLIC icuuc)
add_library(icu INTERFACE)
target_link_libraries(icu INTERFACE icui18n)
add_library(ICU::uc ALIAS icuuc)
add_library(ICU::i18n ALIAS icui18n)
add_library(ICU::data ALIAS icudata)
